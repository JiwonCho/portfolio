'use client';

import { motion, useMotionValue, useSpring, Variants } from 'framer-motion';
import { MouseEvent } from 'react';

// 텍스트 애니메이션용 컨테이너 및 자식 (글자) Variants
const containerVariants: Variants = {
  hidden: { opacity: 0 },
  visible: (i = 1) => ({
    opacity: 1,
    transition: { staggerChildren: 0.04, delayChildren: 0.2 * i },
  }),
};

const childVariants: Variants = {
  hidden: {
    opacity: 0,
    y: 40,
    transition: {
      type: 'spring',
      damping: 12,
      stiffness: 100,
    },
  },
  visible: {
    opacity: 1,
    y: 0,
    transition: {
      type: 'spring',
      damping: 12,
      stiffness: 100,
    },
  },
};

interface TextRevealProps {
  text: string;
  className?: string;
}
```

// 글자 단위 스플릿 및 애니메이션 컴포넌트

```
function TextReveal({ text, className }: TextRevealProps) {
  const letters = Array.from(text);

  return (
    <motion.div
      className={`inline-flex flex-wrap ${className}`}
      variants={containerVariants}
      initial="hidden"
      animate="visible"
    >
      {letters.map((letter, index) => (
        <motion.span key={index} variants={childVariants}>
          {letter === ' ' ? '\u00A0' : letter}
        </motion.span>
      ))}
    </motion.div>
  );
}

export default function HeroSection() {
  // 마우스 좌표 관리를 위한 MotionValue
  const mouseX = useMotionValue(0);
  const mouseY = useMotionValue(0);

  // 반응 속도를 부드럽게 만들어주는 Spring 옵션
  const springConfig = { damping: 25, stiffness: 150 };
  const smoothX = useSpring(mouseX, springConfig);
  const smoothY = useSpring(mouseY, springConfig);

  // 마우스 이동 시 좌표 업데이트
  const handleMouseMove = (e: MouseEvent<HTMLDivElement>) => {
    const { clientX, clientY } = e;
    const { innerWidth, innerHeight } = window;

    // 화면 중앙을 (0,0)으로 위치 정규화 (-0.5 ~ 0.5)
    mouseX.set((clientX / innerWidth) - 0.5);
    mouseY.set((clientY / innerHeight) - 0.5);
  };

  return (
    <section
      onMouseMove={handleMouseMove}
      className="relative flex min-h-screen w-full flex-col items-center justify-center overflow-hidden bg-slate-950 px-6 text-white"
    >
      {/* 1. 마우스 트래킹 3D 패럴랙스 배경 요소 */}
      <motion.div
        style={{
          x: smoothX.get() ? useSpring(smoothX, springConfig) : 0,
          y: smoothY.get() ? useSpring(smoothY, springConfig) : 0,
        }}
        className="pointer-events-none absolute inset-0 flex items-center justify-center"
      >
        {/* 마우스를 따라 부드럽게 움직이는 은은한 글로우 효과 */}
        <motion.div
